━━━ Premiers pas ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  1. Attendre qu'Ollama ait fini de télécharger les modèles (peut prendre 10-15 min) :
     kubectl logs -n {{ .Release.Namespace }} \
       -l app.kubernetes.io/component=ollama -f

  2. Ingérer un document test :
     curl -X POST http://localhost:8001/ingest \
       -F "file=@mon-document.pdf" \
       -F "collection=documents"

  3. Poser une question :
     curl -X POST http://localhost:8000/query \
       -H "Content-Type: application/json" \
       -d '{"question": "Quel est le sujet de ce document ?", "collection": "documents"}'

  4. Lancer les tests Helm :
     helm test {{ .Release.Name }} -n {{ .Release.Namespace }}

━━━ Gestion ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Mise à jour : helm upgrade {{ .Release.Name }} ./helm/rag-system -f my-values.yaml
  Désinstaller : helm uninstall {{ .Release.Name }} -n {{ .Release.Namespace }}

  ⚠️  Les PersistentVolumeClaims (données Qdrant, Ollama, PostgreSQL) ne sont
      PAS supprimés par helm uninstall. Pour tout nettoyer :
      kubectl delete pvc --all -n {{ .Release.Namespace }}
